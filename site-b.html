-- 1. All Trips across 3 regions (via DB Links)
SELECT * FROM Trips                      -- North (local)
UNION
SELECT * FROM Trips@south_db             -- South (remote)
UNION
SELECT * FROM Trips@east_db;              -- East (remote)
 
-- 2. Full itinerary distributed join
SELECT
  b.BookingID,
  t.NAME        AS Tourist,
  b.Amount,
  tr.REGION,
  tr.STARTDATE,
  tr.ENDDATE
FROM Bookings b
JOIN Tourist_Basic t  ON b.TouristID = t.TouristID
JOIN Trips tr         ON b.TripID = tr.TripID
ORDER BY b.BookingID;
 
-- 3. Create Global View
CREATE VIEW All_Trips AS
  SELECT *, 'North' AS Source FROM Trips
  UNION
  SELECT *, 'South' AS Source FROM Trips@south_db
  UNION
  SELECT *, 'East'  AS Source FROM Trips@east_db;
